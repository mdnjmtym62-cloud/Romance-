    // Gestion du système de date (1 message par jour débloqué séquentiellement)
    function getStoredData() {
      const data = localStorage.getItem('prunelle_bocal_data');
      if (data) {
        return JSON.parse(data);
      }
      return {
        unlockedCount: 0, // Nombre de messages débloqués au total (de 0 à 29)
        lastOpenDate: ""  // Date de la dernière ouverture (format "YYYY-MM-DD")
      };
    }

    function saveStoredData(data) {
      localStorage.setItem('prunelle_bocal_data', JSON.stringify(data));
    }

    function getTodayString() {
      const d = new Date();
      return d.toISOString().split('T')[0];
    }

    function showView(viewId) {
      document.querySelectorAll('.view-section').forEach(el => el.classList.remove('active'));
      document.getElementById(viewId).classList.add('active');
      if (viewId === 'archive-view') {
        renderArchive();
      }
    }

    function openTodayMessage() {
      const state = getStoredData();
      const today = getTodayString();

      // Si la personne a déjà ouvert un message aujourd'hui
      if (state.lastOpenDate === today) {
        showView('locked-view');
        return;
      }

      // Si tous les 30 jours ont été ouverts
      if (state.unlockedCount >= poems30Days.length) {
        alert("Tu as déjà ouvert tous les 30 messages de ce bocal ! Tu peux les relire dans ton journal de bord.");
        showView('archive-view');
        return;
      }

      // Débloquer le message du jour
      const currentDayIndex = state.unlockedCount;
      state.unlockedCount += 1;
      state.lastOpenDate = today;
      saveStoredData(state);

      // Afficher le message
      document.getElementById('dayTag').innerText = `JOUR ${state.unlockedCount} / 30`;
      document.getElementById('dailyMessageText').innerText = poems30Days[currentDayIndex];

      // Lancer des confettis magiques !
      confetti({
        particleCount: 70,
        spread: 60,
        origin: { y: 0.6 },
        colors: ['#5c1428', '#d4af37', '#ffffff']
      });

      showView('message-view');
    }

    function renderArchive() {
      const state = getStoredData();
      const container = document.getElementById('archiveContainer');
      container.innerHTML = '';

      if (state.unlockedCount === 0) {
        container.innerHTML = '<p style="font-family: \'Cormorant Garamond\', serif; font-style: italic; color: var(--text-muted); text-align: center; padding: 20px;">Ton bocal est encore intact. Ouvre ton premier mot du jour !</p>';
        return;
      }

      for (let i = 0; i < state.unlockedCount; i++) {
        const item = document.createElement('div');
        item.className = 'archive-item';
        item.innerHTML = `<span>Jour ${i + 1}</span>${poems30Days[i]}`;
        container.appendChild(item);
      }
    }
